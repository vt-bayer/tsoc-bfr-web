Documentação

DAX / pseudocódigo

Regras de negócio do TSOC BFR — espelho do motor do app, no formato Power BI.

IMPLEMENTAÇÃO POWER BI

Realizado (ACT / PY)

Realizado =
-- Relatorio: coluna ACT no ano corrente; PY no ano histórico
CALCULATE(
    SUM( Relatorio[valor_k] ),
    Relatorio[serie] = IF( [AnoSelecionado] = [AnoCorrente], "ACT", "PY" ),
    Relatorio[mes] IN [MesesAteFiltro]
)

Previsto (cenário)

Previsto =
-- Cenário: Budget / FC1 / FC2 / FC3 (ou PY do ano anterior no modo YoY)
VAR cenario = SELECTEDVALUE( Filtros[cenario], "FC2" )
RETURN
CALCULATE(
    SUM( Relatorio[valor_k] ),
    Relatorio[serie] = cenario,
    Relatorio[mes] IN [MesesAteFiltro]
)

Diferença e situação

Diferença = [Realizado] - [Previsto]

Situação =
VAR pct = DIVIDE( ABS( [Diferença] ), ABS( [Previsto] ) )
RETURN
SWITCH(
    TRUE(),
    pct < 0.01, "Dentro do previsto",
    [Diferença] > 0, "Acima do previsto",
    "Economia"
)

Período (mês vs YTD)

MesesAteFiltro =
VAR mes = SELECTEDVALUE( Filtros[mes], "ago" )
VAR modo = SELECTEDVALUE( Filtros[periodo], "ytd" )
RETURN
IF(
    modo = "mes",
    { mes },
    FILTER( VALUES( Calendario[mes] ), Calendario[ordem] <= LOOKUPVALUE( Calendario[ordem], Calendario[mes], mes ) )
)

Sem ACT ≠ economia

DiferençaMensal =
-- Meses sem realizado não geram "economia" falsa no gráfico
VAR real = [RealizadoMes]
VAR prev = [PrevistoMes]
RETURN
IF( ABS( real ) < 0.5 && ABS( prev ) >= 0.5, BLANK(), real - prev )

Filtro sem custos de pessoas

EscopoArea =
-- Preferência do usuário (Personalizar filtros)
CALCULATE(
    [Diferença],
    NOT Relatorio[tipo_de_conta] IN {
        "Benefícios", "Custos com Pessoal", "Encargos",
        "Estágio", "Hora extra", "Indenização",
        "Salários", "STI - PLR", "Temporários"
    }
)
